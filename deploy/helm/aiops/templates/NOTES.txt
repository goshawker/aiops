{{- else if eq (.Values.web.service.type | default "ClusterIP") "NodePort" }}

Get the web UI URL:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services web)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT

{{- else }}

Access the web UI via port-forward:
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/web 8080:{{ .Values.web.service.port }}
  Then open http://localhost:8080
